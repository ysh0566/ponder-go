package abi

import (
	"encoding/json"
	"fmt"
	"go/format"
	"go/token"

	"reflect"

	"strings"
	"unicode"

// GoFieldName maps Solidity component names to exported Go fields. Generation
// rejects collisions; it never silently merges distinct component names.
func GoFieldName(name string) string {
	var out strings.Builder
	upper := true
	for _, r := range name {
		if !unicode.IsLetter(r) && !unicode.IsDigit(r) {
			upper = true
			continue
		}
		if out.Len() == 0 && unicode.IsDigit(r) {
			out.WriteString("Field")
		}
		if upper {
			r = unicode.ToUpper(r)
		}
		out.WriteRune(r)
		upper = false
	}
	if out.Len() == 0 {
		return "Field"
	}
	return out.String()
}

func parameterType(p Parameter) (string, error) {
	if m := fixedArray.FindStringSubmatch(p.Type); m != nil {
		n, e := strconv.ParseUint(m[2], 10, 32)
		if e != nil || n > 1<<20 {
			return "", fmt.Errorf("unsupported fixed ABI array length %q", m[2])
		}
		p.Type = m[1]
		element, e := parameterType(p)
		return fmt.Sprintf("[%d]%s", n, element), e
	}
	if m := dynamicArray.FindStringSubmatch(p.Type); m != nil {
		p.Type = m[1]
		element, e := parameterType(p)
		return "[]" + element, e
	}
	if p.Type == "tuple" {
		named := len(p.Components) > 0
		for _, c := range p.Components {
			named = named && (c.HasName || c.Name != "")
		}
		return parametersType(p.Components, named)
	}
	if p.Type == "string" || p.Type == "address" || p.Type == "bytes" {
		return "string", nil
	}
	if p.Type == "bool" {
		return "bool", nil
	}
	if strings.HasPrefix(p.Type, "bytes") {
		n, e := strconv.Atoi(strings.TrimPrefix(p.Type, "bytes"))
		if e == nil && n >= 1 && n <= 32 {
			return "string", nil
		}
	}
	for _, prefix := range []string{"uint", "int"} {
		if strings.HasPrefix(p.Type, prefix) {
			n := 256
			var e error
			if p.Type != prefix {
				n, e = strconv.Atoi(strings.TrimPrefix(p.Type, prefix))
			}
			if e == nil && n >= 8 && n <= 256 && n%8 == 0 {
				if n > 48 {
					return "*big.Int", nil
				}
				return "int64", nil
			}
		}
	}
	return "", fmt.Errorf("unsupported ABI type %q", p.Type)
}
func parametersType(params []Parameter, named bool) (string, error) {
	var out strings.Builder
	out.WriteString("struct {\n")
	seen := map[string]bool{}
	for i, p := range params {
		field := fmt.Sprintf("Value%d", i)
		if named {
			field = GoFieldName(p.Name)
		}
		if seen[field] {
			return "", fmt.Errorf("ABI component names collide at Go field %q", field)
		}
		seen[field] = true
		typ, e := parameterType(p)
		if e != nil {
			return "", e
		}
		fmt.Fprintf(&out, "%s %s", field, typ)
		if named {
			fmt.Fprintf(&out, " %s", strconv.Quote("abi:"+strconv.Quote(p.Name)))
		}
		out.WriteString("\n")
	}
	out.WriteString("}")
	return out.String(), nil
}

// GenerateParameters produces a native typed decoder from ABI inputs. Tuple
// positions become Value0, Value1, etc.; fixed arrays retain their length, and
// named tuple fields retain their original names through abi struct tags.
// Go has no readonly tuples or template-literal string types; Address/Hex are
// represented by strings, with decoding responsible for their value format.
func GenerateParameters(packageName, name string, params []Parameter) ([]byte, error) {
	if !token.IsIdentifier(packageName) || token.Lookup(packageName).IsKeyword() || packageName == "_" || !token.IsIdentifier(name) || !token.IsExported(name) || token.Lookup(name).IsKeyword() {
		return nil, fmt.Errorf("valid package and exported decoder names required")
	}
	typ, e := parametersType(params, false)
	if e != nil {
		return nil, e
	}
	raw, e := json.Marshal(params)
	if e != nil {
		return nil, e
	}
	var out strings.Builder
	fmt.Fprintf(&out, "// Code generated by ponder-go ABI codegen; DO NOT EDIT.\npackage %s\nimport (\"encoding/json\";\"ponder.local/ponder-go/abi\"", packageName)
	if strings.Contains(typ, "*big.Int") {
		out.WriteString(";\"math/big\"")
	}
	out.WriteString(")\n")
	fmt.Fprintf(&out, "type %sResult = %s\n", name, typ)
	fmt.Fprintf(&out, "var parameters%s = func() []abi.Parameter { var params []abi.Parameter; if err := json.Unmarshal([]byte(%s), &params); err != nil { panic(err) }; return params }()\n", name, strconv.Quote(string(raw)))
	fmt.Fprintf(&out, "func Decode%s(data string, options ...abi.DecodeOptions) (%sResult, error) { var result %sResult; err := abi.DecodeParametersInto(parameters%s, data, &result, options...); return result, err }\n", name, name, name, name)
	return format.Source([]byte(out.String()))
}

// DecodeParametersInto populates the structs/arrays emitted by GenerateParameters.
// A decoding or shape error leaves the caller's destination unchanged.
func DecodeParametersInto(params []Parameter, data string, destination any, options ...DecodeOptions) error {
	target := reflect.ValueOf(destination)
	if target.Kind() != reflect.Pointer || target.IsNil() {
		return fmt.Errorf("ABI destination must be a non-nil pointer")
	}
	expected, e := parametersType(params, false)
	if e != nil {
		return e
	}
	_ = expected // Validate the schema even when the data is empty.
	values, e := DecodeParameters(params, data, options...)
	if e != nil {
		return e
	}
	temporary := reflect.New(target.Elem().Type()).Elem()
	if e = assignDecoded(temporary, values); e != nil {
		return e
	}
	target.Elem().Set(temporary)
	return nil
}

var bigIntType = reflect.TypeOf((*big.Int)(nil))

func assignDecoded(dst reflect.Value, value any) error {
	if !dst.CanSet() {
		return fmt.Errorf("ABI output field is not exported")
	}
	if dst.Type() == bigIntType {
		n, ok := value.(*big.Int)
		if !ok {
			return fmt.Errorf("ABI output requires *big.Int, got %T", value)
		}
		dst.Set(reflect.ValueOf(new(big.Int).Set(n)))
		return nil
	}
	switch dst.Kind() {
	case reflect.Struct:
		if values, ok := value.([]any); ok {
			if len(values) != dst.NumField() {
				return fmt.Errorf("ABI tuple length mismatch")
			}
			for i, v := range values {
				if e := assignDecoded(dst.Field(i), v); e != nil {
					return e
				}
			}
			return nil
		}
		if values, ok := value.(map[string]any); ok {
			if len(values) != dst.NumField() {
				return fmt.Errorf("ABI named tuple length mismatch")
			}
			for i := 0; i < dst.NumField(); i++ {
				key, present := dst.Type().Field(i).Tag.Lookup("abi")
				if !present {
					return fmt.Errorf("ABI named tuple field is missing abi tag")
				}
				v, present := values[key]
				if !present {
					return fmt.Errorf("ABI tuple component %q missing", key)
				}
				if e := assignDecoded(dst.Field(i), v); e != nil {
					return e
				}
			}
			return nil
		}
	case reflect.Array, reflect.Slice:
		values, ok := value.([]any)
		if !ok {
			break
		}
		if dst.Kind() == reflect.Array && len(values) != dst.Len() {
			return fmt.Errorf("ABI fixed array length mismatch")
		}
		if dst.Kind() == reflect.Slice {
			dst.Set(reflect.MakeSlice(dst.Type(), len(values), len(values)))
		}
		for i, v := range values {
			if e := assignDecoded(dst.Index(i), v); e != nil {
				return e
			}
		}
		return nil
	case reflect.String:
		if v, ok := value.(string); ok {
			dst.SetString(v)
			return nil
		}
	case reflect.Int64:
		if v, ok := value.(int64); ok {
			dst.SetInt(v)
			return nil
		}
	case reflect.Bool:
		if v, ok := value.(bool); ok {
			dst.SetBool(v)
			return nil
		}
	}
	return fmt.Errorf("ABI decoded value %T does not match %s", value, dst.Type())
}
