type m43LifetimeRecorder struct { mu sync.Mutex; rows []m43LifetimeRow }
func m43ErrorLeaves(err error) []string {
 var out []string
 var visit func(error,int)
 visit=func(e error,depth int){
  if e==nil { return }; if depth>64 { out=append(out,"ERROR_UNWRAP_DEPTH_LIMIT"); return }
  if many,ok:=e.(interface{Unwrap()[]error}); ok { for _,x:=range many.Unwrap(){visit(x,depth+1)};return }
  if one,ok:=e.(interface{Unwrap()error});ok { if x:=one.Unwrap();x!=nil { visit(x,depth+1);return } }
  out=append(out,fmt.Sprintf("%T: %s",e,e.Error()))
 }
 visit(err,0);return out
}

// The four exact family keys must all be held by one actual connection. No
// basename matching, backend getter, or guessed PID is accepted.
func m43OwnerPID(ctx context.Context,db *sql.DB,ns string)(int,int,error){
 const q=`WITH wanted(k) AS (
 SELECT hashtextextended(x || ':independent-owner',0)
 FROM unnest(ARRAY[$1,$1||'_sync',$1||'_legacy',$1||'_rpc_1']) AS x
 ) SELECT l.pid,count(*) FROM pg_locks l JOIN wanted w
 ON l.classid::bigint=((w.k >> 32)&4294967295)
 AND l.objid::bigint=(w.k&4294967295)
 WHERE l.locktype='advisory' AND l.objsubid=1 AND l.granted
 GROUP BY l.pid ORDER BY l.pid`
 rows,err:=db.QueryContext(ctx,q,ns);if err!=nil{return 0,0,err};defer rows.Close()
 pid,count,n:=0,0,0
 for rows.Next(){n++;if err=rows.Scan(&pid,&count);err!=nil{return 0,0,err}}
 if err=rows.Err();err!=nil{return 0,0,err};if n>1{return 0,0,fmt.Errorf("owner family split across %d PIDs",n)}
 return pid,count,nil
}

func m43ObserveBlockedInsert(ctx context.Context,db *sql.DB,syncPID,lockPID int,table string)(map[string]any,error){
 polls:=0
 for {
  polls++
  var state,waitType,wait,query string;var blocked bool
  err:=db.QueryRowContext(ctx,`SELECT state,coalesce(wait_event_type,''),coalesce(wait_event,''),query,$2=ANY(pg_blocking_pids(pid)) FROM pg_stat_activity WHERE pid=$1`,syncPID,lockPID).Scan(&state,&waitType,&wait,&query,&blocked)
  if err!=nil{return nil,err}
  if blocked&&waitType=="Lock"&&strings.Contains(query,"INSERT INTO "+table){return map[string]any{"sync_pid":syncPID,"blocker_pid":lockPID,"state":state,"wait_event_type":waitType,"wait_event":wait,"query":query,"polls":polls},nil}
  if err=ctx.Err();err!=nil{return nil,fmt.Errorf("no actual blocked cache INSERT observed: %w",err)}
  runtime.Gosched() // Observation polling only; no elapsed-time success predicate.
 }
}

// Observer-owned cancellation callbacks are joined independently of production
// futures. A bounded join failure is reported, never treated as a complete trace.
type m43CancelWatch struct{ done chan struct{}; stop func() bool }
type m43CancelObservers struct{ mu sync.Mutex; sealed bool; watches []*m43CancelWatch; errs []error }
func (o *m43CancelObservers) watch(ctx context.Context,fn func())error{
 o.mu.Lock();defer o.mu.Unlock();if o.sealed{err:=errors.New("observer registration after seal");o.errs=append(o.errs,err);return err}
 w:=&m43CancelWatch{done:make(chan struct{})};o.watches=append(o.watches,w)
 w.stop=context.AfterFunc(ctx,func(){defer close(w.done);fn()});return nil
}
func (o *m43CancelObservers) join(ctx context.Context)error{
 o.mu.Lock();o.sealed=true;items:=append([]*m43CancelWatch(nil),o.watches...);errs:=append([]error(nil),o.errs...);o.mu.Unlock()
 for _,w:=range items{select{case<-w.done:continue;default:};select{case<-w.done:case<-ctx.Done():
  errs=append(errs,fmt.Errorf("cancellation observer did not join: %w",ctx.Err()))
  if w.stop(){close(w.done)} // Prevent an unstarted observer from escaping cleanup.
 }}
 return errors.Join(errs...)
}
func m43OwnerReleased(ctx context.Context,db *sql.DB,businessNS,syncNS string,businessPID,syncPID int)(map[string]any,error){
 polls:=0
 for{polls++;bp,bn,be:=m43OwnerPID(ctx,db,businessNS);sp,sn,se:=m43OwnerPID(ctx,db,syncNS)
  var live int;pe:=db.QueryRowContext(ctx,"SELECT count(*) FROM pg_stat_activity WHERE pid IN ($1,$2)",businessPID,syncPID).Scan(&live)
  row:=map[string]any{"polls":polls,"business_pid":bp,"business_locks":bn,"sync_pid":sp,"sync_locks":sn,"original_pids_remaining":live}
  if err:=errors.Join(be,se,pe);err!=nil{return row,err}
  if bn==0&&sn==0&&live==0{return row,nil}
  if err:=ctx.Err();err!=nil{return row,err};runtime.Gosched()
 }
}
